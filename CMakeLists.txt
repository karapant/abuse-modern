cmake_minimum_required(VERSION 3.24)
project(AbuseModern VERSION 0.1.0 LANGUAGES C CXX)

# The inherited filesystem and networking code currently requires POSIX.
if(NOT UNIX)
    message(FATAL_ERROR "This first port supports Linux and macOS. Windows needs a POSIX/networking port.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ABUSE_FETCH_DEPS "Download pinned SDL dependencies when not installed" ON)
set(ABUSE_ASSET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/data" CACHE PATH "Default external Abuse game-data directory")
include(cmake/Dependencies.cmake)
include(cmake/Sources.cmake)
include(CheckIncludeFile)
foreach(header dirent.h fcntl.h malloc.h memory.h netinet/in.h stdint.h stdlib.h string.h strings.h sys/ioctl.h sys/time.h unistd.h)
    string(TOUPPER "${header}" feature)
    string(REGEX REPLACE "[/ .]" "_" feature "${feature}")
    check_include_file("${header}" HAVE_${feature})
endforeach()
configure_file(cmake/config.h.in generated/config.h)
add_library(abuse_options INTERFACE)
target_include_directories(abuse_options INTERFACE
    "${CMAKE_CURRENT_BINARY_DIR}/generated" src src/imlib src/lisp src/net src/sdlport)
target_compile_definitions(abuse_options INTERFACE HAVE_CONFIG_H NO_CHECK "ASSETDIR=\"${ABUSE_ASSET_DIR}\"")

foreach(component lisp platform imlib net)
    string(TOUPPER "${component}" upper)
    add_library(abuse_${component} STATIC ${ABUSE_${upper}_SOURCES})
    target_link_libraries(abuse_${component} PUBLIC abuse_options)
endforeach()
# The inherited Lisp interpreter represents nil by calling Eval/Print on a null
# LObject. Its copying collector also aliases object and root-pointer types.
# Preserve that legacy behavior until the interpreter's object model is replaced.
target_compile_options(abuse_lisp PUBLIC
    "$<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang,AppleClang>:-fno-delete-null-pointer-checks;-fno-strict-aliasing>")
target_link_libraries(abuse_platform PUBLIC SDL3::SDL3 SDL3_mixer::SDL3_mixer)
add_library(abuse_game OBJECT ${ABUSE_GAME_SOURCES})
target_link_libraries(abuse_game PUBLIC abuse_lisp abuse_platform abuse_imlib abuse_net)
add_executable(abuse src/main.cpp)
target_link_libraries(abuse PRIVATE abuse_game)
add_executable(abuse-tool src/tool/abuse-tool.cpp src/crc.cpp)
target_link_libraries(abuse-tool PRIVATE abuse_imlib)

include(CTest)

    foreach(manager apt dnf pacman zypper brew)

        ENVIRONMENT "ABUSE_SAVE_PATH=${CMAKE_CURRENT_BINARY_DIR}/test-user;ABUSE_PATH=")
